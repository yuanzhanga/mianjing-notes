### 进程通常有几种状态，如何转换

![image-20260830020001203](https://18403581474.oss-cn-beijing.aliyuncs.com/img/202608300200270.png)

```js
function calculatingScore(text) {
    let s = 'coder'
    let arr = text.split(' ').map(i => i.toLowerCase())
    let res = 0
    for (let i = 0; i < arr.length; i++) {
        if (arr[i] === s) res++
    }
    return res
}

module.exports = {
    calculatingScore: calculatingScore
};
```
