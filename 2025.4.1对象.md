# 对象方法

```js
let obj = {}
Object.create(a)//指定a为原型
Object.assign()//浅拷贝 同{...obj}
Object.keys, values, entries
Object.fromEntries()//数组转对象
Object.defineProperty(a, prop, attr)
Object.defineProperties(a, { prop: attr })
Object.getOwnPropertyNames(obj)//获取所有属性名
Object.getOwnPropertySymbols(obj)
Object.getOwnPropertyDescriptor(obj, prop)//获取属性描述符
Object.getOwnPropertyDescriptors(obj)//获取所有属性描述符
Object.is()//是否相等
Object.freeze()
Object.isFrozen()
Object.setPrototypeOf()//设置原型__proto__
Object.getPrototypeOf()//获取原型

obj.hasOwnProperty()//是否有该属性
obj.isPrototypeOf()//是否是原型链上的属性
```
