##### 手写防抖节流

```js
// 节流
function throttle(fn, timeout) {
  let timer = null;
  return function (...arg) {
    if (timer) return
    timer = setTimeout(() => {
      //fn(...arg)
      fn.apply(this, arg)
      timer = null
    }, timeout);
  }
}
//防抖
function debounce(fn, timeout) {
  let timer = null;
  return function (...arg) {
    clearTimeout(timer)
    timer = setTimeout(() => {
      fn.apply(this, arg)

    }, timeout);
  }
}
```
